cmake_minimum_required(VERSION 3.16)
project(hei_chassis_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools geometry_msgs nav_msgs tf2_msgs)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Threads REQUIRED)
add_library(hei_chassis_core SHARED src/protocol.cpp src/transport.cpp)
target_include_directories(hei_chassis_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hei_chassis_core Threads::Threads)
add_library(hei_chassis_plugins SHARED src/controller.cpp src/hardware.cpp)
target_include_directories(hei_chassis_plugins PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hei_chassis_plugins hei_chassis_core Threads::Threads)
ament_target_dependencies(hei_chassis_plugins controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools geometry_msgs nav_msgs tf2_msgs)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
pluginlib_export_plugin_description_file(hardware_interface hardware_plugins.xml)
install(TARGETS hei_chassis_core hei_chassis_plugins EXPORT export_hei_chassis LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config urdf DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/chassis_keyboard.py DESTINATION lib/${PROJECT_NAME} RENAME chassis_keyboard)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_keyboard test/test_keyboard.py)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core hei_chassis_core)
  ament_add_gtest(test_controller test/test_controller.cpp)
  target_link_libraries(test_controller hei_chassis_plugins)
  ament_target_dependencies(test_controller controller_interface hardware_interface rclcpp geometry_msgs)
  ament_add_gtest(test_hardware test/test_hardware.cpp)
  target_link_libraries(test_hardware hei_chassis_plugins util)
  ament_target_dependencies(test_hardware hardware_interface rclcpp)
endif()
ament_export_targets(export_hei_chassis HAS_LIBRARY_TARGET)
ament_export_dependencies(controller_interface hardware_interface pluginlib rclcpp rclcpp_lifecycle realtime_tools geometry_msgs nav_msgs tf2_msgs)
ament_package()
