cmake_minimum_required(VERSION 3.8)
project(hei_robot_mapping)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rtabmap_msgs REQUIRED)
# AGX's NVIDIA OpenCV 4.8 CMake files reference missing runtime libraries.
# Use the deployed Ubuntu 4.5 prefix matching cv_bridge when it is available.
set(_agx_opencv "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/opencv_4_5/prefix/usr/lib/aarch64-linux-gnu/cmake/opencv4")
if(NOT OpenCV_DIR AND EXISTS "${_agx_opencv}/OpenCVConfig.cmake")
  set(OpenCV_DIR "${_agx_opencv}" CACHE PATH "OpenCV matching ROS cv_bridge")
endif()
find_package(RTABMap REQUIRED)
find_package(rtabmap_conversions REQUIRED)
add_library(map_snapshot STATIC src/map_snapshot.cpp)
target_compile_features(map_snapshot PUBLIC cxx_std_17)
target_compile_options(map_snapshot PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(map_snapshot PUBLIC include)
ament_target_dependencies(map_snapshot rtabmap_msgs rtabmap_conversions)
target_link_libraries(map_snapshot rtabmap::core)
add_executable(map_saver src/map_saver.cpp)
target_compile_features(map_saver PUBLIC cxx_std_17)
target_compile_options(map_saver PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(map_saver rclcpp rtabmap_msgs)
target_link_libraries(map_saver map_snapshot)
add_executable(fastlio_cloud_adapter src/fastlio_cloud_adapter.cpp)
target_compile_features(fastlio_cloud_adapter PUBLIC cxx_std_17)
target_compile_options(fastlio_cloud_adapter PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(fastlio_cloud_adapter PRIVATE include)
ament_target_dependencies(fastlio_cloud_adapter rclcpp sensor_msgs)
install(TARGETS fastlio_cloud_adapter map_saver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_xyzi_conversion test/test_xyzi_conversion.cpp)
  target_compile_features(test_xyzi_conversion PUBLIC cxx_std_17)
  target_include_directories(test_xyzi_conversion PRIVATE include)
  ament_target_dependencies(test_xyzi_conversion sensor_msgs)
  ament_add_gtest(test_map_snapshot test/test_map_snapshot.cpp)
  target_compile_features(test_map_snapshot PUBLIC cxx_std_17)
  ament_target_dependencies(test_map_snapshot rtabmap_msgs rtabmap_conversions)
  target_link_libraries(test_map_snapshot map_snapshot)
endif()
ament_package()
