cmake_minimum_required(VERSION 3.8)
project(hei_robot_mapping)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(fastlio_cloud_adapter src/fastlio_cloud_adapter.cpp)
target_compile_features(fastlio_cloud_adapter PUBLIC cxx_std_17)
target_compile_options(fastlio_cloud_adapter PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(fastlio_cloud_adapter PRIVATE include)
ament_target_dependencies(fastlio_cloud_adapter rclcpp sensor_msgs)
install(TARGETS fastlio_cloud_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_xyzi_conversion test/test_xyzi_conversion.cpp)
  target_compile_features(test_xyzi_conversion PUBLIC cxx_std_17)
  target_include_directories(test_xyzi_conversion PRIVATE include)
  ament_target_dependencies(test_xyzi_conversion sensor_msgs)
endif()
ament_package()
